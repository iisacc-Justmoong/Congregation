cmake_minimum_required(VERSION 3.31)
if(NOT CMAKE_BINARY_DIR STREQUAL "${CMAKE_SOURCE_DIR}/build")
    message(FATAL_ERROR "Configure Congregation in the repository build/ directory")
endif()
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum macOS version")
endif()
project(Congregation VERSION 1.0.0 LANGUAGES C CXX)
if(MSVC)
    add_compile_options(/utf-8 /permissive-)
endif()
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTORCC OFF)
set(CMAKE_AUTOUIC OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt5 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Qt6 ON)
set(CONGREGATION_SDK_PREFIX "${CMAKE_SOURCE_DIR}/../../SDK/build/congregation-native/prefix" CACHE PATH "Installed native iiSDK kit")
list(PREPEND CMAKE_PREFIX_PATH "${CONGREGATION_SDK_PREFIX}")
foreach(_package iiSDKCore iiJson iiPaintEngine iiFileProviderNative iiFileProviderCoreIO iiFileProviderCompatibility iiNetwork iiAcountManagerNative iiLicenseManagerNative iiUpdateManagerNative iiSharedCanvasNative iiImageCodec LVUI iiGUI LVRS lunasvg plutovg)
    if(EXISTS "${CONGREGATION_SDK_PREFIX}/lib/cmake/${_package}")
        set(${_package}_DIR "${CONGREGATION_SDK_PREFIX}/lib/cmake/${_package}" CACHE PATH "Native kit package" FORCE)
    endif()
endforeach()
include(CTest)
option(CONGREGATION_BUILD_GUI "Build the LVUI/LVRS native application" ON)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
find_package(iiSDKCore 0.18 CONFIG REQUIRED COMPONENTS Runtime)
find_package(iiJson 0.1 CONFIG REQUIRED)
find_package(iiFileProviderNative 0.9 CONFIG REQUIRED)
find_package(iiFileProviderCoreIO CONFIG REQUIRED)
find_package(iiFileProviderCompatibility 0.9 CONFIG REQUIRED)
find_package(iiNetwork 0.5 CONFIG REQUIRED)
find_package(iiAcountManagerNative CONFIG REQUIRED)
find_package(iiLicenseManagerNative CONFIG REQUIRED)
find_package(iiUpdateManagerNative CONFIG REQUIRED)
find_package(iiSharedCanvasNative 0.32 CONFIG REQUIRED)
find_package(iiImageCodec 0.1 CONFIG REQUIRED)
find_package(LibXml2 REQUIRED)
add_subdirectory(App/Engine Engine)
find_package(iiGUI 0.4 CONFIG REQUIRED)
add_library(CongregationArtwork STATIC App/Native/Artwork.cpp App/Native/Artwork.h App/Native/Shapes.cpp App/Native/LegacyCanvas.cpp)
target_compile_features(CongregationArtwork PUBLIC cxx_std_23)
target_include_directories(CongregationArtwork PUBLIC App)
target_compile_definitions(CongregationArtwork PUBLIC CONGREGATION_NATIVE_WIDGET_ABI="${iiGUI_NATIVE_ABI_ID}")
target_link_libraries(CongregationArtwork PUBLIC Congregation::Engine iiSharedCanvas::Native PRIVATE iiGUI::iiGUI)
if(CONGREGATION_BUILD_GUI)
    find_package(LVRS 1.0 CONFIG REQUIRED)
    find_package(LVUI 0.1 CONFIG REQUIRED)
    find_package(iiGUI 0.4 CONFIG REQUIRED)
    set(_editor_assets "")
    foreach(_icon IN ITEMS panHand translateObject addFile generalopen generalsave generalsearch showCode eraser rectangle ellipse triangle diamond star fillbucket typeAlias imagezoomOut imagezoomIn)
        file(READ "${CMAKE_SOURCE_DIR}/resources/icons/${_icon}.svg" _svg)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "resources/icons/${_icon}.svg")
        string(APPEND _editor_assets "{\"${_icon}\",R\"SVG(${_svg})SVG\"},\n")
    endforeach()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/EditorAssets.generated.h" CONTENT "#pragma once\n#include <string_view>\nnamespace congregation::native::assets { struct Icon { std::string_view name,svg; }; inline constexpr Icon icons[] = {${_editor_assets}}; }\n")
    lvui_generate(NAME MainView SOURCE App/Native/views/Main.lvui OUTPUT_VARIABLE _main_view FRAMEWORK LVRS INCLUDES Native/EditorBackend.h)
    lvui_generate(NAME PreferencesView SOURCE App/Native/views/Preferences.lvui OUTPUT_VARIABLE _preferences_view FRAMEWORK LVRS INCLUDES Native/EditorBackend.h)
    lvui_generate(NAME CanvasSetupView SOURCE App/Native/views/CanvasSetup.lvui OUTPUT_VARIABLE _setup_view FRAMEWORK LVRS INCLUDES Native/EditorBackend.h)
    iisdkcore_add_application(Congregation GUI SOURCES App/Native/Application.cpp App/Native/EditorIcons.cpp App/Native/EditorIcons.h App/Native/EditorLayout.cpp App/Native/EditorLayout.h App/Native/CanvasView.cpp App/Native/CanvasView.h App/Native/EditorBackend.cpp App/Native/EditorBackend.h "${_main_view}" "${_preferences_view}" "${_setup_view}"
        LIBRARIES CongregationArtwork LVRS::LVRS iiGUI::iiGUI RESOURCES ${iiGUI_NATIVE_RESOURCES} "${CMAKE_SOURCE_DIR}/resources/Appicon.icns")
    target_include_directories(Congregation PRIVATE App "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_features(Congregation PRIVATE cxx_std_23)
    target_compile_definitions(Congregation PRIVATE CONGREGATION_VERSION="${PROJECT_VERSION}")
    set_target_properties(Congregation PROPERTIES CXX_EXTENSIONS OFF MACOSX_BUNDLE_GUI_IDENTIFIER com.iisacc.congregation.painter MACOSX_BUNDLE_BUNDLE_NAME Congregation MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}" MACOSX_BUNDLE_BUNDLE_VERSION 1 MACOSX_BUNDLE_ICON_FILE Appicon.icns)
    if(APPLE)
        set_target_properties(Congregation PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/cmake/NativeInfo.plist.in" BUILD_RPATH "@executable_path/../Frameworks;${CONGREGATION_SDK_PREFIX}/lib")
        add_custom_command(TARGET Congregation POST_BUILD COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/cmake/package_native_macos.py" --bundle "$<TARGET_BUNDLE_DIR:Congregation>" --prefix "${CONGREGATION_SDK_PREFIX}" VERBATIM)
    endif()
endif()
include(cmake/VerifyNativeEngine.cmake)
if(BUILD_TESTING)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/native-tests")
    add_test(NAME Congregation.native_build_surface COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_SOURCE_DIR}/tests/verify_native_build_surface.py" "${CMAKE_SOURCE_DIR}")
    add_test(NAME Congregation.native_build_surface_contract COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_SOURCE_DIR}/tests/test_native_build_surface.py" "${CMAKE_BINARY_DIR}/native-tests")
    add_executable(tests_applicationengine tests/tst_applicationengine.cpp)
    target_link_libraries(tests_applicationengine PRIVATE Congregation::Engine)
    add_test(NAME Congregation.engine COMMAND tests_applicationengine)
    add_executable(tests_nativebackends tests/tst_nativebackends.cpp)
    target_link_libraries(tests_nativebackends PRIVATE Congregation::Engine)
    add_test(NAME Congregation.services COMMAND tests_nativebackends "${CMAKE_BINARY_DIR}/native-tests")
    add_executable(tests_nativeartwork tests/tst_nativeartwork.cpp)
    target_link_libraries(tests_nativeartwork PRIVATE CongregationArtwork)
    add_test(NAME Congregation.artwork COMMAND tests_nativeartwork "${CMAKE_BINARY_DIR}/native-tests")
    add_executable(tests_editorlayout tests/tst_editorlayout.cpp App/Native/EditorLayout.cpp)
    target_include_directories(tests_editorlayout PRIVATE App)
    target_link_libraries(tests_editorlayout PRIVATE iiGUI::iiGUI)
    add_test(NAME Congregation.editor_layout COMMAND tests_editorlayout)
    if(CONGREGATION_BUILD_GUI)
        foreach(renderer native vulkan)
            set(_renderer_argument "${renderer}")
            if(renderer STREQUAL "native" AND NOT APPLE AND NOT WIN32)
                set(_renderer_argument auto)
            endif()
            add_test(NAME Congregation.standalone_${renderer} COMMAND Congregation --self-test "${CMAKE_BINARY_DIR}/native-tests" --renderer ${_renderer_argument})
            add_test(NAME Congregation.vincent_ui_${renderer} COMMAND Congregation --self-test "${CMAKE_BINARY_DIR}/native-tests" --renderer ${_renderer_argument} --ui-reference "${CMAKE_SOURCE_DIR}/docs/marketing/congregation-sample-artwork.png")
            set_tests_properties(Congregation.vincent_ui_${renderer} PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE)
            set_tests_properties(Congregation.standalone_${renderer} PROPERTIES TIMEOUT 300 RUN_SERIAL TRUE)
        endforeach()
        if(APPLE)
            add_test(NAME Congregation.bundle_relocation COMMAND "${Python3_EXECUTABLE}" -B
                "${CMAKE_SOURCE_DIR}/tests/verify_standalone_bundle.py"
                --bundle "$<TARGET_BUNDLE_DIR:Congregation>" --build "${CMAKE_BINARY_DIR}")
            set_tests_properties(Congregation.bundle_relocation PROPERTIES TIMEOUT 750 RUN_SERIAL TRUE)
        endif()
    endif()
    get_property(_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${_tests} PROPERTIES ENVIRONMENT "TMPDIR=${CMAKE_BINARY_DIR}/native-tests;TMP=${CMAKE_BINARY_DIR}/native-tests;TEMP=${CMAKE_BINARY_DIR}/native-tests")
    set_tests_properties(${_tests} PROPERTIES ENVIRONMENT_MODIFICATION "DYLD_LIBRARY_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;DYLD_FALLBACK_FRAMEWORK_PATH=unset:;DYLD_INSERT_LIBRARIES=unset:;LD_LIBRARY_PATH=unset:")
endif()
